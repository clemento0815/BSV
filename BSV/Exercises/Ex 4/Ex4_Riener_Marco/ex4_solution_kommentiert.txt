# Import standard packages

import matplotlib.pyplot as plt
import pydicom
from skimage.filters import sobel

if __name__ == '__main__':
    
    # Daten einlesen
    # Clemens: Hier werden noch keine Daten eingelesen, sondern nur die
    #          Dateipfade festgelegt. Vielleicht "Dateipfade festlegen"?
    in_file1 = 'SER00002/2A1A1ADC'
    in_file2 = 'SER00003/0B8C096A'
    # Clemens: Bei mir läuft das Programm leider nicht, es kommt
    #          "FileNotFoundError: No such file or directory". Ich glaube, der
    #          Pfad funktioniert nur, wenn man das Programm genau aus deinem
    #          Ordner startet. Man könnte den Pfad mit "import os" und
    #          os.path.join(os.path.dirname(os.path.abspath(__file__)), ...)
    #          zusammenbauen, dann geht es von überall.
    
    # Daten Variablen zuweisen
    data1 = pydicom.dcmread(in_file1)
    data2 = pydicom.dcmread(in_file2)
    
    # Daten zuweisen (Bilddatei)
    img_data1 = data1.pixel_array
    img_data2 = data2.pixel_array
    
    # Clemens: In der Aufgabe soll man bestimmen, welche Ansicht und welche
    #          Einstellung (T1, T2, PD) die Daten haben. Die Ansicht steht nur
    #          im Titel, aber man sieht nicht, woher man das weiß, und die
    #          Einstellung fehlt ganz. Ich glaube, man kann das mit
    #          print(data1.ProtocolName) aus den Metadaten auslesen.
    
    # Sobel-Filter auf eines der Datasets anwenden
    edges = sobel(img_data2)
    
    # Plotten des ersten Bildes
    plt.title("Koronale Ansicht")
    plt.imshow(img_data1, cmap = 'gray')
    plt.show()
    # Clemens: Wenn ich das richtig verstehe, muss man dieses Fenster zuerst
    #          schließen, bevor der Rest vom Programm weiterläuft. Ist das so
    #          gewollt?
    
    # Eine gemeinsame Abbildung erstellen (1 Zeile, 2 Spalten, Bildgröße: B = 12, H = 6)
    fig, axes = plt.subplots(1, 2, figsize=(12, 6))
    
    # Erstes Subplot: Originales Bild, axes[0] = erstes Bilder der Zeile
    # Clemens: Kleiner Tippfehler: "erstes Bild" statt "erstes Bilder".
    axes[0].imshow(img_data2, cmap='gray')
    axes[0].set_title("Original (Sagittale Ansicht)")
    axes[0].axis('off')
    
    # Zweites Subplot: Kantenbild (Sobel), axes[1] = zweites Bild der Zeile
    axes[1].imshow(edges, cmap='gray')
    axes[1].set_title("Kanten (Sobel-Filter)")
    axes[1].axis('off')
    
    # Layout anpassen
    plt.tight_layout()
    
    # Bild abspeichern (vor dem plotten)
    # Clemens: Das Bild wird in dem Ordner gespeichert, aus dem man das
    #          Programm startet, nicht neben dem Skript. Gleiches Problem wie
    #          oben bei den Pfaden.
    plt.savefig('knee_edges.jpg', dpi=300)
    
    #Ausgabe
    plt.show()
